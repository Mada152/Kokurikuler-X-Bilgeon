function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); // Menambahkan baris baru ke spreadsheet sheet.appendRow([ data.id || "", data.tanggal || "", data.hari || "", data.jam || "", data.jp || "", data.kelas || "", data.nama || "", data.keluhan || "", data.dicatatPada || new Date() ]); return ContentService.createTextOutput(JSON.stringify({"result": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
